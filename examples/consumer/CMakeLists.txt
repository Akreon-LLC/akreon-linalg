cmake_minimum_required(VERSION 3.30)
project(AkreonLinalgConsumer LANGUAGES CXX)
option(AKREON_LINALG_ENABLE_SANITIZERS "Instrument the consumer example with ASan and UBSan" OFF)
find_package(AkreonLinalg 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Akreon::Linalg)
if(AKREON_LINALG_ENABLE_SANITIZERS)
  target_link_libraries(consumer PRIVATE Akreon::LinalgSanitizers)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
